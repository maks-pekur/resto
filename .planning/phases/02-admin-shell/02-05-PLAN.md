    ```typescript
    'use server';

    import { z } from 'zod';
    import pino from 'pino';
    import { apiFetch } from '@/lib/api-server';

    const EmailSchema = z.object({ email: z.string().email().max(255) });

    export interface NotifyListSignupState {
      readonly ok: boolean;
      readonly error: string | null;
    }

    const logger = pino({ name: 'admin.notify-list-signup' });

    // CONTEXT D-17 degraded path: weak in-process throttle to prevent log-flood DoS
    // when the real api endpoint is still 404. The REAL rate limit arrives with
    // the Phase 03 api endpoint; this is a stopgap, not a security boundary.
    const THROTTLE_WINDOW_MS = 60_000;
    const THROTTLE_MAX = 5;
    const recentSignups: { ts: number; email: string }[] = [];

    const shouldThrottle = (email: string): boolean => {
      const now = Date.now();
      // prune entries outside the window
      while (recentSignups.length > 0 && now - recentSignups[0]!.ts > THROTTLE_WINDOW_MS) {
        recentSignups.shift();
      }
      const distinctEmails = new Set(recentSignups.map((s) => s.email));
      if (distinctEmails.size >= THROTTLE_MAX && !distinctEmails.has(email)) {
        return true;
      }
      recentSignups.push({ ts: now, email });
      return false;
    };

    export async function notifyListSignupAction(
      _prev: NotifyListSignupState,
      formData: FormData,
    ): Promise<NotifyListSignupState> {
      const parsed = EmailSchema.safeParse({ email: formData.get('email') });
      if (!parsed.success) {
        return { ok: false, error: 'Please enter a valid email.' };
      }
      const { email } = parsed.data;
      const res = await apiFetch<{ ok?: boolean }>('/v1/marketing/notify-list', {
        method: 'POST',
        body: { email },
      });
      if (res.status === 404) {
        // CONTEXT D-17: degrade gracefully when the backend endpoint isn't wired yet.
        // Capture the email server-side; Phase 03+ batch-resubscribes via the planned endpoint.
        if (!shouldThrottle(email)) {
          logger.info({ email }, 'ai_notify_list_signup (backend not yet wired)');
        }
        return { ok: true, error: null };
      }
      if (res.ok) {
        return { ok: true, error: null };
      }
      if (res.status >= 400 && res.status < 500) {
        return { ok: false, error: 'We could not add you to the list. Try again in a moment.' };
      }
      return { ok: false, error: 'Something went wrong. Please try again.' };
    }
    ```

    'use client';

    import { useActionState } from 'react';
    import { useFormStatus } from 'react-dom';
    import { Loader2 } from 'lucide-react';
    import { Button } from '@/components/ui/button';
    import { Input } from '@/components/ui/input';

              {state.error}
            </p>
          ) : null}
        </form>
      );
    }
