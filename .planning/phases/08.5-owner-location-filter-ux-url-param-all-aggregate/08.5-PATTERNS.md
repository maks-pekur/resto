```typescript
@Injectable()
export class OwnerOnlyGuard implements CanActivate {
  constructor(@Inject(Reflector) private readonly reflector: Reflector) {}

  canActivate(ctx: ExecutionContext): boolean {
    const required = this.reflector.getAllAndOverride<boolean | undefined>(
      OWNER_ONLY_KEY,
      [ctx.getHandler(), ctx.getClass()],
    );
    if (!required) return true;

    const req = ctx.switchToHttp().getRequest<FastifyRequest>();
    const principal = req.principal;
    if (principal?.kind !== 'operator' || principal.baseRole !== 'owner') {
      throw new NotFoundException();
    }
    return true;
  }
}
```

Note: unlike `LocationScopeGuard`/`PermissionsGuard`, this guard has no `await` — keep it synchronous (`canActivate(): boolean`, not `Promise<boolean>`) since no repository lookup is needed; NestJS accepts both.

```typescript
async listStopListAggregateAcrossLocations(
  tenantId: TenantId,
  activeLocationIds: readonly string[],
): Promise<AggregateStopListRow[]> {
  if (activeLocationIds.length === 0) return [];
  return this.db.withTenant(async (tx) => {
    const grouped = await tx
      .select({
        itemId: schema.menuStopList.itemId,
        stoppedLocationCount: sql<number>`count(distinct ${schema.menuStopList.locationId})`,
        lastStoppedAt: sql<Date>`max(${schema.menuStopList.stoppedAt})`,
      })
      .from(schema.menuStopList)
      .where(
        and(
          eq(schema.menuStopList.tenantId, tenantId),
          inArray(schema.menuStopList.locationId, activeLocationIds),
        ),
      )
      .groupBy(schema.menuStopList.itemId)
      .orderBy(desc(sql`max(${schema.menuStopList.stoppedAt})`))
      .limit(50)
      .offset(0);
    // batch-fetch item/category names exactly like listStopListWithStoppedAt (Analog A) does
    return grouped;
  });
}
```

`sql` is already imported at the top of this file (line 20: `import { and, asc, desc, eq, ilike, inArray, isNull, ne, or, sql } from 'drizzle-orm';`) — no new import needed. No new migration required: `menu_stop_list_location_item_tenant_uq (tenant_id, location_id, item_id)` (migration `0068_stop_list_location_regrain.sql`) already serves this `WHERE tenant_id = X AND location_id IN (...)` shape.
