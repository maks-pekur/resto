### `packages/domain/src/rbac/non-delegatable.ts` (utility, transform) — NEW FILE

**Analog:** `packages/domain/src/rbac/permissions.ts` (same package, same shape)

**Imports pattern** (mirror permissions.ts line 1):

```typescript

```

**Core pattern** (from RESEARCH.md §D-04):

```typescript
export const NON_DELEGATABLE: Permission = {
  tenant: ['delete', 'transfer'],
  billing: ['update'],
  staff: ['remove'],
  ac: ['create', 'read', 'update', 'delete'],
} as const;

export function containsNonDelegatable(
  permission: Record<string, string[]>,
): boolean {
  for (const [resource, actions] of Object.entries(NON_DELEGATABLE) as [
    keyof typeof NON_DELEGATABLE,
    string[],
  ][]) {
    const requested = permission[resource] ?? [];
    if (actions.some((a) => requested.includes(a))) return true;
  }
  return false;
}
```

**Core pattern** (mirror bootstrap-owner.service.ts `execute` + BA API call pattern at lines 66–152):

```typescript
export interface CreateRoleInput {
  readonly roleName: string;
  readonly permission: Record<string, string[]>;
  readonly organizationId: string;    // always from ALS, never from client
  readonly actorUserId: string;
  readonly headers: Headers;          // forwarded to BA for session context
}

@Injectable()
export class CreateRoleService {
  private readonly logger = new Logger(CreateRoleService.name);

  constructor(
    @Inject(AUTH_TOKEN) private readonly auth: Auth,
    @Inject(IDENTITY_EVENT_EMITTER) private readonly emitter: IdentityEventEmitterPort,
  ) {}

  async execute(input: CreateRoleInput): Promise<{ roleSlug: string }> {
    // 1. NON_DELEGATABLE guard — MUST run before calling BA
    if (containsNonDelegatable(input.permission)) {
      throw new InsufficientPermissionsToMintError();
    }
    // 2. Call BA via typed cast (same pattern as permission-checker.adapter.ts lines 36-46)
    const api = this.auth.api as unknown as { createOrgRole: ... };
    const result = await api.createOrgRole({
      body: { organizationId: input.organizationId, role: input.roleName, permission: input.permission },
      headers: input.headers,
    });
    // 3. Emit audit event (mirror auth.config.ts lines 232-258)
    const tenantId = TenantId.parse(input.organizationId);
    await this.emitter.emit(buildEnvelope(IdentityRoleCreatedV1, { ... }, { tenantId }));
    return { roleSlug: result.role ?? input.roleName };
  }
}
```

private async lookupBaseRole(

  const rows = await this.authDb.db
    .select({ role: memberTable.role })
