```typescript
export const pkUuid = () =>
  uuid('id')
    .primaryKey()
    .default(sql`gen_random_uuid()`);

export const tenantIdColumn = () => uuid('tenant_id').notNull();

export const timestampsColumns = () => ({
  createdAt: timestamp('created_at', { withTimezone: true, mode: 'date' })
    .notNull()
    .default(sql`now()`),
  updatedAt: timestamp('updated_at', { withTimezone: true, mode: 'date' })
    .notNull()
    .default(sql`now()`),
  archivedAt: timestamp('archived_at', { withTimezone: true, mode: 'date' }),
});

export const tenantParentUniqueIndex = (
  tableName: string,
  cols: { id: PgColumn; tenantId: PgColumn },
) => uniqueIndex(`${tableName}_id_tenant_uq`).on(cols.id, cols.tenantId);

export const compositeTenantFk = (params: {
  name: string;
  child: { id: PgColumn; tenantId: PgColumn };
  parent: { id: PgColumn; tenantId: PgColumn };
}) =>
  foreignKey({
    name: params.name,
    columns: [params.child.id, params.child.tenantId],
    foreignColumns: [params.parent.id, params.parent.tenantId],
  });
```

```sql
CREATE OR REPLACE FUNCTION app_bind_location(p_location TEXT)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_current TEXT := current_setting('app.current_location', true);
BEGIN
  IF v_current IS NOT NULL AND v_current <> '' AND v_current <> p_location THEN
    RAISE EXCEPTION
      'app.current_location already bound to % — refusing to rebind to %',
      v_current, p_location

END

DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'resto_app') THEN
    EXECUTE 'GRANT EXECUTE ON FUNCTION app_bind_location(text) TO resto_app';
  END IF;
END
$$;
```
